      self,

      Uses the Phoenix transformer to encode user + history embeddings
      into a single user representation vector.
      """
      # 1. Embed user features
      user_embeddings, user_padding_mask = block_user_reduce(
          batch.user_hashes,
          recsys_embeddings.user_embeddings,
          hash_config.num_user_hashes,
          config.emb_size,
          1.0,
      )

      # 2. Embed engagement history (posts + authors + actions)
      history_embeddings, history_padding_mask = block_history_reduce(
          batch.history_post_hashes,
          recsys_embeddings.history_post_embeddings,
          recsys_embeddings.history_author_embeddings,
          history_product_surface_embeddings,
          history_actions_embeddings,  # ← LIKES, REPLIES, REPOSTS, etc.
          hash_config.num_item_hashes,
          hash_config.num_author_hashes,

      )

      # 3. Concatenate user + history
      embeddings = jnp.concatenate([user_embeddings, history_embeddings],
  axis=1)
      padding_mask = jnp.concatenate([user_padding_mask, history_padding_mask],
  axis=1)

      # 4. Pass through Grok transformer
      model_output = self.model(  # ← GROK TRANSFORMER
          embeddings.astype(self.fprop_dtype),
          padding_mask,
          candidate_start_offset=None,

      # 5. Pool transformer outputs → single user vector
      user_outputs = model_output.embeddings
      mask_float = padding_mask.astype(jnp.float32)[:, :, None]
      user_embeddings_masked = user_outputs * mask_float
      user_embedding_sum = jnp.sum(user_embeddings_masked, axis=1)  # Sum 
  pooling
      mask_sum = jnp.sum(mask_float, axis=1)
      user_representation = user_embedding_sum / jnp.maximum(mask_sum, 1.0)  # 
  Average

      # 6. L2 normalize for dot product similarity
      user_norm_sq = jnp.sum(user_representation**2, axis=-1, keepdims=True)
      user_norm = jnp.sqrt(jnp.maximum(user_norm_sq, EPS))
      user_representation = user_representation / user_norm  # ← L2 NORMALIZED

      return user_representation, user_norm

  def make_recsys_attn_mask(
      seq_len: int,
      candidate_start_offset: int,
      dtype: jnp.dtype = jnp.float32,
  ) -> jax.Array:
      """Create attention mask for recommendation system inference.
      
      Creates a mask where:
      - Positions 0 to candidate_start_offset-1 (user+history): causal attention
      - Positions candidate_start_offset onwards (candidates): can attend to 
  user+history
        and themselves (self-attention), but NOT to other candidates
      
      This ensures each candidate is scored independently based on user+history 
  context.
      """
      # Start with causal mask for the full sequence
      causal_mask = jnp.tril(jnp.ones((1, 1, seq_len, seq_len), dtype=dtype))

      # Zero out candidate-to-candidate attention (bottom-right block)
      attn_mask = causal_mask.at[:, :, candidate_start_offset:,
  candidate_start_offset:].set(0)

      # Add back self-attention for candidates (diagonal of the candidate block)
      candidate_indices = jnp.arange(candidate_start_offset, seq_len)
      attn_mask = attn_mask.at[:, :, candidate_indices,
  candidate_indices].set(1)

      return attn_mask

  Attention Mask Visualization:
  Sequence: [User, History_1, History_2, ..., Candidate_1, Candidate_2, ...]

  Attention Matrix:
                     U   H1  H2  ...  C1  C2  C3
  User               1   0   0   ...  0   0   0   ← Causal
  History_1          1   1   0   ...  0   0   0   ← Causal
  History_2          1   1   1   ...  0   0   0   ← Causal
  ...
  Candidate_1        1   1   1   ...  1   0   0   ← Can see user+history, only
  self
  Candidate_2        1   1   1   ...  0   1   0   ← Can see user+history, only
  self
  Candidate_3        1   1   1   ...  0   0   1   ← Can see user+history, only
  self

  Why This Matters:
  Without isolation:
    score(C1) = f(user, history, C2, C3)  ← Depends on batch composition

  With isolation:
    score(C1) = f(user, history, C1)      ← Independent, cacheable
    score(C2) = f(user, history, C2)
    score(C3) = f(user, history, C3)

  🎯 YES - This Creates a "Cold Start Problem" for New Accounts

  The Network Effect Penalty

  From the code in oon_scorer.rs, for ANY post:

  Visibility Formula:
    - To your F followers: score = base_score × 1.0 (no penalty)
    - To everyone else: score = base_score × OON_WEIGHT_FACTOR (< 1.0)

  For a NEW account (F ≈ 0):
    - Nearly all potential viewers get penalized scores
    - Even if content quality is high, visibility is suppressed

  Mathematical Reach Model

  Expected Reach = F × P(show | in_network) + (N - F) × P(show | OON)

  Where:
    F = follower count
    N = total active users
    P(show | in_network) = probability of showing in feed (no penalty)
    P(show | OON) = P(show | in_network) × OON_WEIGHT_FACTOR

  Simplifying:
    Reach = F × P + (N - F) × P × OON_WEIGHT_FACTOR
          = P × [F + (N - F) × OON_WEIGHT_FACTOR]
          = P × [F × (1 - OON_WEIGHT_FACTOR) + N × OON_WEIGHT_FACTOR]

  As F → 0 (new account):
  Reach_new ≈ N × P × OON_WEIGHT_FACTOR

  As F → N (popular account):
  Reach_popular ≈ N × P × 1.0

  Reach Ratio:
  Reach_popular / Reach_new = 1.0 / OON_WEIGHT_FACTOR

  If OON_WEIGHT_FACTOR = 0.5:
    Popular accounts get 2x the reach of new accounts

  If OON_WEIGHT_FACTOR = 0.7:
    Popular accounts get ~1.43x the reach

  Cons:
    ✗ New accounts disadvantaged
    ✗ Network effects amplify inequality
    ✗ Harder to discover new creators
    ✗ Entrenches popular accounts

  def build_user_representation(
      self,
      batch: RecsysBatch,
      recsys_embeddings: RecsysEmbeddings,
  ) -> Tuple[jax.Array, jax.Array]:
      """Build user representation from user features and history.
      
      Uses the Phoenix transformer to encode user + history embeddings
      into a single user representation vector.
      """
      # 1. Embed user features
      user_embeddings, user_padding_mask = block_user_reduce(
          batch.user_hashes,
          recsys_embeddings.user_embeddings,
          hash_config.num_user_hashes,
          config.emb_size,
          1.0,
      )

      # 2. Embed engagement history (posts + authors + actions)
      history_embeddings, history_padding_mask = block_history_reduce(
          batch.history_post_hashes,
          recsys_embeddings.history_post_embeddings,
          recsys_embeddings.history_author_embeddings,
          history_product_surface_embeddings,
          history_actions_embeddings,  # ← LIKES, REPLIES, REPOSTS, etc.
          hash_config.num_item_hashes,
          hash_config.num_author_hashes,
          1.0,
      )

      # 3. Concatenate user + history
      embeddings = jnp.concatenate([user_embeddings, history_embeddings],
  axis=1)
      padding_mask = jnp.concatenate([user_padding_mask, history_padding_mask],
  axis=1)

      # 4. Pass through Grok transformer
      model_output = self.model(  # ← GROK TRANSFORMER
          embeddings.astype(self.fprop_dtype),
          padding_mask,
          candidate_start_offset=None,
      )

      # 5. Pool transformer outputs → single user vector
      user_outputs = model_output.embeddings
      mask_float = padding_mask.astype(jnp.float32)[:, :, None]
      user_embeddings_masked = user_outputs * mask_float
      user_embedding_sum = jnp.sum(user_embeddings_masked, axis=1)  # Sum 
  pooling
      mask_sum = jnp.sum(mask_float, axis=1)
      user_representation = user_embedding_sum / jnp.maximum(mask_sum, 1.0)  # 
  Average

      # 6. L2 normalize for dot product similarity
      user_norm_sq = jnp.sum(user_representation**2, axis=-1, keepdims=True)
      user_norm = jnp.sqrt(jnp.maximum(user_norm_sq, EPS))
      user_representation = user_representation / user_norm  # ← L2 NORMALIZED

      return user_representation, user_norm

  CandidateTower MLP:
  def __call__(self, post_author_embedding: jax.Array) -> jax.Array:
      # Flatten hash embeddings
      post_author_embedding = jnp.reshape(post_author_embedding, (B, C, -1))

      # Two-layer MLP with SiLU activation
      hidden = jnp.dot(post_author_embedding, proj_1)  # [B, C, emb_size*2]
      hidden = jax.nn.silu(hidden)  # ← SiLU activation
      candidate_embeddings = jnp.dot(hidden, proj_2)  # [B, C, emb_size]

      # L2 normalize
      candidate_norm = jnp.sqrt(jnp.sum(candidate_embeddings**2, axis=-1,
  keepdims=True))
      candidate_representation = candidate_embeddings / candidate_norm

      return candidate_representation

  3C. Retrieval via Dot Product Similarity

  def _retrieve_top_k(
      self,
      user_representation: jax.Array,  # [B, D]
      corpus_embeddings: jax.Array,    # [N, D] (N = billions of tweets)
      top_k: int,
      corpus_mask: Optional[jax.Array] = None,
  ) -> Tuple[jax.Array, jax.Array]:
      """Retrieve top-k candidates from a corpus for each user."""

      # Dot product similarity (both are L2 normalized, so this is cosine 
  similarity)
      scores = jnp.matmul(user_representation, corpus_embeddings.T)  # [B, N]

      if corpus_mask is not None:
          scores = jnp.where(corpus_mask[None, :], scores, -INF)

      # Select top-K
      top_k_scores, top_k_indices = jax.lax.top_k(scores, top_k)  # ← TOP-K 
  SELECTION

      return top_k_indices, top_k_scores

  Mathematical Formula:
  similarity(user, candidate) = user_embedding · candidate_embedding
                               = ||user|| ||candidate|| cos(θ)
                               = cos(θ)  (since both are L2-normalized)

  score ∈ [-1, 1] where:
    1.0 = perfect alignment
   -1.0 = opposite direction

  def make_recsys_attn_mask(
      seq_len: int,
      candidate_start_offset: int,
      dtype: jnp.dtype = jnp.float32,
  ) -> jax.Array:
      """Create attention mask for recommendation system inference.
      
      Creates a mask where:
      - Positions 0 to candidate_start_offset-1 (user+history): causal attention
      - Positions candidate_start_offset onwards (candidates): can attend to 
  user+history
        and themselves (self-attention), but NOT to other candidates
      
      This ensures each candidate is scored independently based on user+history 
  context.
      """
      # Start with causal mask for the full sequence
      causal_mask = jnp.tril(jnp.ones((1, 1, seq_len, seq_len), dtype=dtype))

      # Zero out candidate-to-candidate attention (bottom-right block)
      attn_mask = causal_mask.at[:, :, candidate_start_offset:,
  candidate_start_offset:].set(0)

      # Add back self-attention for candidates (diagonal of the candidate block)
      candidate_indices = jnp.arange(candidate_start_offset, seq_len)
      attn_mask = attn_mask.at[:, :, candidate_indices,
  candidate_indices].set(1)

      return attn_mask

  Attention Mask Visualization:
  Sequence: [User, History_1, History_2, ..., Candidate_1, Candidate_2, ...]

  Attention Matrix:
                     U   H1  H2  ...  C1  C2  C3
  User               1   0   0   ...  0   0   0   ← Causal
  History_1          1   1   0   ...  0   0   0   ← Causal
  History_2          1   1   1   ...  0   0   0   ← Causal
  ...
  Candidate_1        1   1   1   ...  1   0   0   ← Can see user+history, only
  self
  Candidate_2        1   1   1   ...  0   1   0   ← Can see user+history, only
  self
  Candidate_3        1   1   1   ...  0   0   1   ← Can see user+history, only
  self

  Why This Matters:
  Without isolation:
    score(C1) = f(user, history, C2, C3)  ← Depends on batch composition

  With isolation:
    score(C1) = f(user, history, C1)      ← Independent, cacheable
    score(C2) = f(user, history, C2)
